{{- /* Renders a 1200x630 sharing card for the given page and returns the image resource, or nothing if the pieces are missing. Called from templates/_funcs/get-page-images.html as the last fallback, so both the Open Graph and the Twitter card tags pick it up. Hugo caches the result under resources/_gen, so a card is drawn once per title and reused on later builds. */ -}} {{- $base := resources.Get "images/og-base.png" }} {{- $logo := resources.Get "images/og-logo.png" }} {{- $font := resources.Get "fonts/LiberationSans-Bold.ttf" }} {{- $size := 62 }} {{- $leading := 20 }} {{- $maxLines := 4 }} {{- $maxWidth := 30 }} {{- /* The text block lives between the logo and the byline. */ -}} {{- $areaTop := 185 }} {{- $areaBottom := 505 }} {{- $out := "" }} {{- if and $base $font }} {{- /* On the home page the page title is the site title, so a card built from it would print the name twice — once large, once in the byline. Use the site description there instead: it says what the blog is. */ -}} {{- $title := .Title | plainify }} {{- if and .IsHome site.Params.description }} {{- $title = site.Params.description | plainify }} {{- end }} {{- /* images.Text draws lines but does not break them, so break by hand. Then even the lines out: a greedy break fills each line to the brim and leaves a stub behind — "Claude Code" alone on a fourth line. Re-breaking at the narrowest width that still needs the same number of lines spreads the words across them instead. */ -}} {{- $lines := partial "wrap_text.html" (dict "text" $title "width" $maxWidth) }} {{- $n := len $lines }} {{- if gt $n 1 }} {{- $floor := math.Ceil (div (strings.RuneCount $title) $n) | int }} {{- if and (ge $floor 1) (lt $floor $maxWidth) }} {{- range seq $floor $maxWidth }} {{- $try := partial "wrap_text.html" (dict "text" $title "width" .) }} {{- if eq (len $try) $n }} {{- $lines = $try }} {{- break }} {{- end }} {{- end }} {{- end }} {{- end }} {{- /* Longer than the card holds; the last kept line gets an ellipsis. */ -}} {{- if gt (len $lines) $maxLines }} {{- $lines = (first (sub $maxLines 1) $lines) | append (printf "%s …" (index $lines (sub $maxLines 1))) }} {{- end }} {{- /* Centre the block vertically, so a two-line title leaves no hole under it and a four-line one does not crowd the byline. */ -}} {{- $count := len $lines }} {{- $blockHeight := add (mul $count $size) (mul (sub $count 1) $leading) }} {{- $y := add $areaTop (div (sub (sub $areaBottom $areaTop) $blockHeight) 2) }} {{- $img := $base }} {{- with $logo }} {{- $img = $img.Filter (images.Overlay . 80 64) }} {{- end }} {{- $img = $img.Filter (images.Text (delimit $lines "\n") (dict "font" $font "size" $size "linespacing" $leading "color" "#f4f4f5" "x" 80 "y" $y)) }} {{- $img = $img.Filter (images.Text (printf "%s · %s" site.Title (.Date.Format "2006")) (dict "font" $font "size" 30 "color" "#9a9aa2" "x" 80 "y" 550)) }} {{- /* Stable name per title and language so the cache actually hits. The name must survive a round trip through the web server. urlize alone percent-encodes an umlaut, which produces a file literally named "...gr%C3%BChn.png"; the server then decodes the request back to "grühn" and answers 404. So transliterate first, then drop anything that is not plain ASCII. */ -}} {{- $slug := .Title | lower }} {{- range $from, $to := (dict "ä" "ae" "ö" "oe" "ü" "ue" "ß" "ss") }} {{- $slug = replace $slug $from $to }} {{- end }} {{- $slug = replaceRE "[^a-z0-9-]" "" ($slug | urlize) }} {{- $out = $img | resources.Copy (printf "og/%s-%s.png" .Lang $slug) }} {{- end }} {{- return $out }}
