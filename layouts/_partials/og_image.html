{{- /* Renders a 1200x630 sharing card for the given page and returns the image resource, or nothing if the pieces are missing. Called from templates/_funcs/get-page-images.html as the last fallback, so both the Open Graph and the Twitter card tags pick it up. Hugo caches the result under resources/_gen, so a card is drawn once per title and reused on later builds. */ -}} {{- $base := resources.Get "images/og-base.png" }} {{- $logo := resources.Get "images/og-logo.png" }} {{- $font := resources.Get "fonts/LiberationSans-Bold.ttf" }} {{- $out := "" }} {{- if and $base $font }} {{- /* Wrap the title by hand: images.Text draws lines but does not break them. Rune count, not byte count, so umlauts do not shorten a line. */ -}} {{- /* On the home page the page title is the site title, so a card built from it would print the name twice — once large, once in the byline. Use the site description there instead: it says what the blog is. */ -}} {{- $title := .Title | plainify }} {{- if and .IsHome site.Params.description }} {{- $title = site.Params.description | plainify }} {{- end }} {{- $lines := slice }} {{- $cur := "" }} {{- range split $title " " }} {{- $word := . }} {{- $try := cond (eq $cur "") $word (printf "%s %s" $cur $word) }} {{- if gt (strings.RuneCount $try) 30 }} {{- if ne $cur "" }}{{ $lines = $lines | append $cur }}{{ end }} {{- $cur = $word }} {{- else }} {{- $cur = $try }} {{- end }} {{- end }} {{- if ne $cur "" }}{{ $lines = $lines | append $cur }}{{ end }} {{- /* Four lines fill the card; anything beyond gets an ellipsis. */ -}} {{- if gt (len $lines) 4 }} {{- $lines = append (printf "%s …" (index $lines 3)) (first 3 $lines) }} {{- end }} {{- $img := $base }} {{- with $logo }} {{- $img = $img.Filter (images.Overlay . 80 64) }} {{- end }} {{- $img = $img.Filter (images.Text (delimit $lines "\n") (dict "font" $font "size" 62 "linespacing" 20 "color" "#f4f4f5" "x" 80 "y" 190)) }} {{- $img = $img.Filter (images.Text (printf "%s · %s" site.Title (.Date.Format "2006")) (dict "font" $font "size" 30 "color" "#9a9aa2" "x" 80 "y" 550)) }} {{- /* Stable name per title and language so the cache actually hits. The name must survive a round trip through the web server. urlize alone percent-encodes an umlaut, which produces a file literally named "...gr%C3%BChn.png"; the server then decodes the request back to "grühn" and answers 404. So transliterate first, then drop anything that is not plain ASCII. */ -}} {{- $slug := .Title | lower }} {{- range $from, $to := (dict "ä" "ae" "ö" "oe" "ü" "ue" "ß" "ss") }} {{- $slug = replace $slug $from $to }} {{- end }} {{- $slug = replaceRE "[^a-z0-9-]" "" ($slug | urlize) }} {{- $out = $img | resources.Copy (printf "og/%s-%s.png" .Lang $slug) }} {{- end }} {{- return $out }}
